## 🔧 使用方法

### 进程池（自动启用）

```rust
use game_engine::scripting::csharp_dotnet::DotNetCliHost;

// 进程池自动启用
let host = DotNetCliHost::initialize()?;

// 查看统计
if let Some(stats) = host.get_process_pool_stats() {
    println!("Hit rate: {}%", stats["hit_rate_percent"]);
}

// 定期维护
host.health_check_process_pool();
host.cleanup_idle_processes();
```

### 热重载（手动启用）

```rust
use game_engine::scripting::csharp::CSharpContext;
use std::path::PathBuf;

let mut ctx = CSharpContext::new();

// 启用热重载
ctx.enable_hot_reload(
    vec![PathBuf::from("./scripts")],
    100  // 100ms 防抖动
)?;

// 主循环
loop {
    // 检查热重载
    if let Ok(reloaded) = ctx.check_hot_reload() {
        if !reloaded.is_empty() {
            println!("Reloaded {} scripts", reloaded.len());
        }
    }

    // 游戏逻辑...
}
```
